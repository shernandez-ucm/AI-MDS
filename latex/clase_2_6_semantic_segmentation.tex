					\item Respuesta: \textit{que es cada parte}.
					\item Salida: mapa de etiquetas (altura $\times$ ancho).
				\end{itemize}
			\end{block}
		\end{column}
	\end{columns}

\section{Arquitecturas Cl\'asicas}

\begin{frame}{U-Net: Arquitectura Fundamental}

	La arquitectura \textbf{U-Net} (Ronneberger et al., 2015) fue disenada para segmentaci\'on medica.

	\formula{
		\[
		\text{Encoder:} \quad H \to H/2 \to H/4 \to \cdots \qquad \text{Decoder:} \quad \cdots \to H/4 \to H/2 \to H
		\]
	}

	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\begin{block}{Ventajas}
				\small
				\begin{itemize}
					\item Pocos par\'ametros (muy eficiente).
					\item Skip connections garantizan detalles.
					\item Excelente para datos pequenos.
				\end{itemize}
			\end{block}
		\end{column}

		\begin{column}{0.5\textwidth}
			\begin{block}{Desventajas}
				\small
				\begin{itemize}
					\item Receptive field limitado.
					\item Sin contexto global expl\'icito.
					\item Requiere mas datos para escenas complejas.
				\end{itemize}
			\end{block}
		\end{column}
	\end{columns}

\end{frame}

\begin{frame}{SegNet: Pooling con \'Indices}

	SegNet (Badrinarayanan et al., 2017) almacena solo los \'indices del max-pooling en el encoder y los reutiliza para el unpooling en el decoder.

\begin{frame}{DeepLab: Contexto Global con Atrous Convolutions}

	\textbf{DeepLab} introduce convoluci\'on atrous (dilatada) para capturar contexto multi-escala:

	\formula{
		\[
		y[i] = \sum_{k} x[i + r \cdot k] \cdot w[k], \quad r = \text{tasa de dilataci\'on}
		\]
	}

	\begin{center}
		\begin{tikzpicture}[scale=0.8]
			\node[draw=mitgray, fill=LightGray, minimum width=3cm, minimum height=0.6cm, font=\small] (img) at (0,2) {Entrada};

		Mayor receptive field sin reducir resoluci\'on. Detalles + contexto global.
	\end{alertblock}

\section{Funciones Objetivo}

\begin{frame}{Dice Loss: Optimizar Solapamiento}

	El \textbf{coeficiente Dice} es una metrica que tambien funciona como perdida:

	\formula{
		\[
		\text{Dice}(y, \hat{y}) = \frac{2 |y \cap \hat{y}|}{|y| + |\hat{y}|}, \qquad \mathcal{L}_{\text{Dice}} = 1 - \text{Dice}
		\]
	}

	\begin{block}{Ventajas sobre Cross-Entropy}
		\begin{itemize}
			\item Directamente optimiza IoU (metrica final).
			\item Menos sensible a desbalance de clases.
			\item Enfatiza objetos pequenos.
		\end{itemize}
	\end{block}

	\begin{alertblock}{Estrategia Practica}
		Combina CE + Dice. Agrega boundary loss si los bordes son criticos.
	\end{alertblock}

\end{frame}

\section{Metricas de Evaluacion}

\begin{frame}{Metricas Pixel-Level}

	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\begin{block}{Dice (F1-Score)}
				\small
				\[
				\text{Dice} = \frac{2\text{TP}}{2\text{TP} + \text{FP} + \text{FN}}
				\]
				Robusto a desbalance de clases.
			\end{block}

			\begin{block}{Sensibilidad (Recall)}
				\small
				\[
				\text{Sens} = \frac{\text{TP}}{\text{TP} + \text{FN}}
				\]
				Proporci\'on de positivos detectados.
			\end{block}
		\end{column}

		\begin{column}{0.5\textwidth}
			\begin{block}{IoU (Jaccard)}
				\small
				\[
				\text{IoU} = \frac{\text{TP}}{\text{TP} + \text{FP} + \text{FN}}
				\]
				Intersecci\'on sobre uni\'on.
			\end{block}

			\begin{block}{Especificidad}
				\small
				\[
				\text{Spec} = \frac{\text{TN}}{\text{TN} + \text{FP}}
				\]
				Tasa de negativos correctos.
			\end{block}
		\end{column}
	\end{columns}

	\vspace{0.2cm}

	\begin{alertblock}{Recomendacion}
		Reporta \textbf{siempre} Dice e IoU, mas sensibilidad y especificidad.
	\end{alertblock}

\end{frame}

\section{Aplicaciones en Oftalmolog\'ia}

\begin{frame}{Retinopat\'ia Diabetica}
\begin{figure}
	\centering
	\includegraphics[width=0.8\textwidth, height=0.5\textheight, 
	keepaspectratio]{figures/diabetic_eye}
\end{figure}
	La \textbf{retinopat\'ia diab\'etica (DR)} es la principal causa de ceguera en adultos desarrollados.

	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\begin{block}{Signos Tempranos}
				\small
				\begin{itemize}
					\item Microaneurismas.
					\item Hemorragias retinianas.
					\item Edema macular diabetico.
				\end{itemize}
			\end{block}
		\end{column}

		\begin{column}{0.5\textwidth}
			\begin{block}{Desafio}
				\small
				\begin{itemize}
					\item Lesiones pequenas y dispersas.
					\item Requiere expertos oftalmologos.
					\item Screening de poblaciones masivas.
				\end{itemize}
			\end{block}
		\end{column}
	\end{columns}


\end{frame}

\begin{frame}{Segmentacion de Vasos Sanguineos}

	Extraer la \textbf{red vascular retiniana} de imagenes de fundus.

	\vspace{0.2cm}

	\begin{block}{Arquitectura tipica}
		\small
		U-Net o SegNet. Entrada: $512 \times 512 \times 3$. Salida: $512 \times 512 \times 1$ (vaso/no-vaso).
	\end{block}

	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\begin{block}{Desafios}
				\small
				\begin{itemize}
					\item Vasos muy finos (1-2 pixeles).
					\item Bajo contraste en areas oscuras.
					\item Precision en bordes.
				\end{itemize}
			\end{block}
		\end{column}

		\begin{column}{0.5\textwidth}
			\begin{block}{Metricas clave}
				\small
				Dice $\geq 0.8$, Sensibilidad $\geq 0.75$.
			\end{block}
		\end{column}
	\end{columns}

\end{frame}

\begin{frame}{Desafios en Oftalmologia}

	\begin{columns}[T]
		\begin{column}{0.33\textwidth}
			\begin{block}{Datos Limitados}
				\small
				Datasets pequenos, anotacion costosa, imbalance de clases.

				\textbf{Solucion:} Transfer learning, augmentation.
			\end{block}
		\end{column}

		\begin{column}{0.33\textwidth}
			\begin{block}{Calidad de Imagen}
				\small
				Ruido, bajo contraste, cataratas, desalineamiento.

				\textbf{Solucion:} Preprocessing (CLAHE).
			\end{block}
		\end{column}

		\begin{column}{0.33\textwidth}
			\begin{block}{Interpretabilidad}
				\small
				Clinicos necesitan explicacion del modelo.

				\textbf{Solucion:} Grad-CAM, attention maps.
			\end{block}
		\end{column}
	\end{columns}

\end{frame}
